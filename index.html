suspend fun fetchData(): Flow<Data> = flow {
  emit(Data.Loading)
  try {
    val result = repository.getData()
    emit(Data.Success(result))
  } catch (e: Exception) {
    emit(Data.Error(e))
  }
}

viewModelScope.launch {
  fetchData().collect { data ->
    updateUi(data)
  }
}
